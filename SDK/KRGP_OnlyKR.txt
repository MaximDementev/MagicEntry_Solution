# Только конструкторы
$searcher = New-Object DirectoryServices.DirectorySearcher
$searcher.Filter = "(&(objectClass=user)(objectCategory=person))"
$searcher.PageSize = 1000

$searcher.PropertiesToLoad.Add("sAMAccountName") | Out-Null
$searcher.PropertiesToLoad.Add("department")     | Out-Null

$users = $searcher.FindAll()

$list = $users | ForEach-Object {
    $user = ($_.Properties["sAMAccountName"] | Select-Object -First 1)
    $dept = ($_.Properties["department"]     | Select-Object -First 1)

    if (-not $dept) { return }
    if ($dept -notlike "КР*") { return }

    [PSCustomObject]@{
        User       = $user
        Department = $dept
    }
}

$totalCount = $list.Count

$output = New-Object System.Text.StringBuilder
[void]$output.AppendLine("Всего конструкторов: $totalCount")
[void]$output.AppendLine("")

$list |
    Group-Object Department |
    Sort-Object Name |
    ForEach-Object {
        [void]$output.AppendLine("[$($_.Name)]")

        $_.Group |
            Sort-Object User |
            ForEach-Object {
                [void]$output.AppendLine($_.User)
            }

        [void]$output.AppendLine("")
    }

$tempFile = [System.IO.Path]::GetTempFileName()
$output.ToString() | Set-Content -Encoding UTF8 $tempFile

notepad.exe $tempFile
