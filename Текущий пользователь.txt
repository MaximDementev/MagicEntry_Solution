Запрос данных текущего пользователя:
using var ctx = new PrincipalContext(ContextType.Domain, null, "OU=Institute,DC=kgp,DC=com");
using var searcher = new PrincipalSearcher(new UserPrincipal(ctx));
foreach (var result in searcher.FindAll())
{
     if (result is UserPrincipal user && !string.IsNullOrEmpty(result.SamAccountName))
     {
         DirectoryEntry de = (DirectoryEntry)user.GetUnderlyingObject();

         if (de.Properties["company"]?.Value?.ToString() is string companyName &&
             !user.DisplayName.Contains("Краснодаргражданпроект") &&
             !companyName.Contains("Краснодаргражданпроект"))
         {
             var dep = de.Properties["department"]?.Value?.ToString() ?? "";
             yield return new AdUserInfo
             {
                 FullName = user.DisplayName ?? "",
                 Email = user.EmailAddress ?? "",
                 Phone = de.Properties["telephoneNumber"]?.Value?.ToString() ?? "",
                 Position = de.Properties["title"]?.Value?.ToString() ?? "",
                 Login = user.SamAccountName,
                 Company = companyName,
                 WorkGroup = dep.Contains("_") ? int.Parse(dep.Substring(dep.IndexOf("_") + 1)) : null,
                 Department = dep
             };
         }
     }
}


Запрос карточки объекта через PowerShell
$domain = "KGP.com"
$username = "dementev"

# Создаем объект для поиска в AD
$searcher = New-Object DirectoryServices.DirectorySearcher
$searcher.Filter = "(&(objectClass=user)(sAMAccountName=$username))"

# Выполняем поиск
$result = $searcher.FindOne()

if ($result) {
    $user = $result.GetDirectoryEntry()
    # Выводим свойства пользователя
    $user | Select-Object *
} else {
    Write-Host "Пользователь не найден"
}
